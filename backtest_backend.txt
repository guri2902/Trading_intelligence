from __future__ import annotations

from .models import CostConfig


def apply_slippage(price: float, direction: str, slippage_bps: float) -> float:
    """
    Buy/long fills are worsened upward; sell/short fills worsened downward.
    """
    factor = slippage_bps / 10_000.0
    if direction in {"LONG", "BUY"}:
        return price * (1.0 + factor)
    if direction in {"SHORT", "SELL"}:
        return price * (1.0 - factor)
    return price


def transaction_cost(
    *,
    entry_value: float,
    exit_value: float,
    config: CostConfig,
) -> float:
    turnover = abs(entry_value) + abs(exit_value)

    brokerage = 2.0 * config.brokerage_per_order
    exchange_cost = turnover * config.exchange_txn_rate
    sebi = turnover * config.sebi_rate
    stamp = abs(entry_value) * config.stamp_duty_rate
    stt = abs(exit_value) * config.stt_rate

    taxable_base = brokerage + exchange_cost + sebi
    gst = taxable_base * config.gst_rate

    return brokerage + exchange_cost + sebi + stamp + stt + gst

from datetime import datetime
from typing import Iterable


def validate_historical_rows(
    rows: list[dict],
    *,
    timestamp_key: str = "timestamp",
) -> list[dict]:
    """
    Produces a chronological, duplicate-free copy.

    Backtests must operate on monotonically increasing historical data.
    """
    normalized = []

    for row in rows:
        ts = row[timestamp_key]
        if isinstance(ts, str):
            ts = datetime.fromisoformat(ts.replace("Z", "+00:00"))
        normalized.append((ts, row))

    normalized.sort(key=lambda x: x[0])

    output = []
    seen = set()

    for ts, row in normalized:
        if ts in seen:
            raise ValueError(f"Duplicate historical timestamp: {ts}")
        seen.add(ts)
        output.append(row)

    if len(output) < 2:
        raise ValueError("At least two historical rows are required")

    return output


def ensure_no_future_data(
    *,
    signal_time: datetime,
    decision_time: datetime,
) -> None:
    # A historical decision may only use information available at or before
    # the signal timestamp. The execution timestamp may legitimately be later.
    if decision_time < signal_time:
        raise ValueError(
            f"Chronology violation: execution/decision time={decision_time} "
            f"< signal time={signal_time}"
        )

import math
import statistics

from .models import BacktestMetrics, EquityPoint, TradeRecord


def _returns(equity: list[EquityPoint]) -> list[float]:
    if len(equity) < 2:
        return []

    out: list[float] = []

    for a, b in zip(equity, equity[1:]):
        if a.equity > 0:
            out.append(b.equity / a.equity - 1.0)

    return out


def sharpe_ratio(equity: list[EquityPoint]) -> float | None:
    r = _returns(equity)

    if len(r) < 2:
        return None

    std = statistics.stdev(r)

    if std == 0:
        return None

    return statistics.mean(r) / std * math.sqrt(len(r))


def sortino_ratio(equity: list[EquityPoint]) -> float | None:
    r = _returns(equity)

    if len(r) < 2:
        return None

    downside = [x for x in r if x < 0]

    if not downside:
        return None

    downside_dev = math.sqrt(
        sum(x * x for x in downside) / len(downside)
    )

    if downside_dev == 0:
        return None

    return statistics.mean(r) / downside_dev * math.sqrt(len(r))


def compute_metrics(
    *,
    initial_capital: float,
    trades: list[TradeRecord],
    equity: list[EquityPoint],
) -> BacktestMetrics:

    # ---------------------------------------------------------
    # REALIZED TRADE P&L
    # ---------------------------------------------------------
    # Trade records are the authoritative source for realized
    # strategy performance. This prevents final equity from
    # becoming inconsistent with gross profit/loss and expectancy.
    trade_pnls = [float(t.net_pnl) for t in trades]

    wins = [pnl for pnl in trade_pnls if pnl > 0]
    losses = [pnl for pnl in trade_pnls if pnl < 0]

    gross_profit = sum(wins)
    gross_loss = abs(sum(losses))

    total = len(trade_pnls)
    winning = len(wins)
    losing = len(losses)

    # Net realized P&L comes directly from the completed trades.
    net_profit = sum(trade_pnls)

    # Final realized equity is therefore deterministic.
    final_equity = initial_capital + net_profit

    # ---------------------------------------------------------
    # BASIC PERFORMANCE
    # ---------------------------------------------------------
    return_pct = (
        (net_profit / initial_capital) * 100.0
        if initial_capital > 0
        else 0.0
    )

    win_rate_pct = (
        (winning / total) * 100.0
        if total
        else 0.0
    )

    profit_factor = (
        gross_profit / gross_loss
        if gross_loss > 0
        else None
    )

    # Mean realized P&L per completed trade.
    expectancy = (
        net_profit / total
        if total
        else 0.0
    )

    # DRAWDOWN
    # ---------------------------------------------------------
    # Drawdown remains based on the equity curve because it is
    # a path-dependent risk metric.
    drawdowns = [

        for point in equity
    ]

    drawdown_pcts = [
        float(point.drawdown_pct)
        for point in equity
    ]

    max_drawdown = (
        max(drawdowns)
        if drawdowns
        else 0.0
    )

    max_drawdown_pct = (
        max(drawdown_pcts)
        if drawdown_pcts
        else 0.0
    )

    # ---------------------------------------------------------
    # CONSISTENCY CHECK
    # ---------------------------------------------------------
    # The metrics must reconcile:
    #
    # final_equity = initial_capital + net_profit
    #
    # and
    #

    #

    expected_net_profit = gross_profit - gross_loss

        ordered = sorted(rows, key=lambda r: r.timestamp)
        if ordered != rows:
            raise ValueError("OPTION_DATA_MUST_BE_CHRONOLOGICALLY_SORTED")
        seen = set()
        for row in rows:
            key = (row.contract_key, row.timestamp)
            if key in seen:
                raise ValueError("DUPLICATE_OPTION_BAR")
            seen.add(key)
            if row.high < max(row.open, row.close) or row.low > min(row.open, row.close):
                raise ValueError("INVALID_OPTION_OHLC")
            if row.bid is not None and row.ask is not None and row.bid > row.ask:
                raise ValueError("BID_ASK_CROSSED")
        return rows

            exchange_txn_rate=config.exchange_txn_rate,
            sebi_rate=config.sebi_rate,
            gst_rate=config.gst_rate,
            stamp_duty_rate=config.stamp_duty_rate,
            stt_rate=config.stt_sell_rate,
        )
        return transaction_cost(entry_value=entry * qty, exit_value=exit * qty, config=cc)

    @staticmethod
    def _buy_fill(row: HistoricalOptionBar, slippage_bps: float) -> float:
        reference = row.ask if row.ask is not None else row.open
        return reference * (1 + slippage_bps / 10000)

    @staticmethod
    def _sell_fill(row: HistoricalOptionBar, slippage_bps: float) -> float:
        reference = row.bid if row.bid is not None else row.open
        return reference * (1 - slippage_bps / 10000)

        max_dd = 0.0

            equity += trade.net_pnl
            peak = max(peak, equity)

                return list(csv.DictReader(fh))
        try:
            import pandas as pd  # type: ignore
        except ImportError as exc:
            raise RuntimeError(
                "OptionVault parquet/feather support requires pandas. "
                "CSV samples work without the optional dependency."
            ) from exc
        if suffix == ".parquet":
            return pd.read_parquet(path).to_dict(orient="records")
        return pd.read_feather(path).to_dict(orient="records")

    def candles(
        self,
        instrument_key,
        unit,
        interval,
        from_date,
        to_date,
    ) -> list[Candle]:
        rows: dict[datetime, Candle] = {}
        candidates = list(self._candidate_files(instrument_key))

        for path in candidates:
            for row in self._read_table(path):
                ts = self._parse_timestamp(
                    self._row_value(row, "datetime", "timestamp", "date")
                )
                if not (from_date <= ts.date() <= to_date):
                    continue
                op = self._row_value(row, "open", "o")
                hi = self._row_value(row, "high", "h")
                lo = self._row_value(row, "low", "l")
                cl = self._row_value(row, "close", "c")
                if None in (op, hi, lo, cl):
                    continue
                vol = self._row_value(row, "volume", "v")
                oi = self._row_value(row, "oi", "open_interest")
                rows[ts] = Candle(
                    instrument_key=instrument_key,
                    timestamp=ts,
                    open=float(op),
                    high=float(hi),
                    low=float(lo),
                    close=float(cl),
                    volume=float(vol) if vol not in (None, "") else None,
                    open_interest=float(oi) if oi not in (None, "") else None,
                    source=self.name,
                )

        return sorted(rows.values(), key=lambda x: x.timestamp)

    def option_series(
        self,
        symbol: str,
        *,
        expiry: date | None = None,
        strike: float | None = None,
        side: str | None = None,
        from_date: date | None = None,
        to_date: date | None = None,
    ) -> list[dict[str, Any]]:
        """Load actual option market rows when the local OptionVault data exists."""
        out: list[dict[str, Any]] = []
        symbol_upper = symbol.upper().replace(" ", "")
        side_upper = side.upper() if side else None

        for path in self._candidate_files(symbol, options=True):
            if symbol_upper not in path.stem.upper().replace(" ", ""):
                continue
            for row in self._read_table(path):
                exp_value = self._row_value(row, "expiry", "exp")
                row_strike = self._row_value(row, "strike", "strike_price")
                row_side = self._row_value(row, "side", "option_type", "type")

                if expiry is not None and exp_value not in (None, ""):
                    try:
                        exp = date.fromisoformat(str(exp_value)[:10])
                    except ValueError:
                        exp = None
                    if exp is not None and exp != expiry:
                        continue
                if strike is not None and row_strike not in (None, ""):
                    if float(row_strike) != float(strike):
                        continue
                if side_upper and row_side not in (None, ""):
                    if str(row_side).upper() != side_upper:
                        continue

                ts = self._parse_timestamp(self._row_value(row, "datetime", "timestamp", "date"))
                if from_date and ts.date() < from_date:
                    continue
                if to_date and ts.date() > to_date:
                    continue

                normalized = {str(k).strip().lower(): v for k, v in row.items()}
                normalized["timestamp"] = ts.isoformat()
                normalized["source"] = self.name
                out.append(normalized)

        out.sort(key=lambda x: x["timestamp"])
        return out

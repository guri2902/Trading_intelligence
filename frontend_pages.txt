"use client";

import { useEffect, useState } from "react";
import { Shell } from "../components/Shell";

const API_BASE =
  process.env.NEXT_PUBLIC_API_BASE_URL || "http://localhost:8000";

type PipelineStatus = {
  running: boolean;
  cycle_count: number;
  last_error: string | null;
  interval_seconds: number | null;
  last_result?: {
    status?: string;
    timestamp?: string;
    candle_timestamp?: string;
    signal?: {
      action?: string;
      direction?: string;
      confidence?: number;
      score?: number;
      hard_gates?: string[];
      selected?: Record<string, unknown> | null;
      live_signal_gate?: {
        passed?: boolean;
        reasons?: string[];
      };
    };
    risk?: {
      approved?: boolean;
      final_action?: string;
      assessment?: {
        reason_codes?: string[];
        risk_budget?: number;
        quantity?: number;
        lots?: number;
        estimated_trade_risk?: number;
        risk_pct_of_equity?: number;
      };
    };
  };
};

export default function Home() {
  const [data, setData] = useState<PipelineStatus | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  async function loadStatus() {
    try {
      const response = await fetch(`${API_BASE}/pipeline/status`, {
        cache: "no-store",
      });

      if (!response.ok) {
        throw new Error(`API returned ${response.status}`);
      }

      const payload = await response.json();
      setData(payload);
      setError("");
    } catch (err) {
      setError(err instanceof Error ? err.message : "Backend unavailable");
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadStatus();

    const timer = setInterval(loadStatus, 5000);

    return () => clearInterval(timer);
  }, []);

  const signal = data?.last_result?.signal;
  const risk = data?.last_result?.risk;

  const action = signal?.action || "WAIT";
  const finalAction = risk?.final_action || "NO_TRADE";

  const dataHealthy =
    data?.running &&
    data?.last_result?.status === "UPDATED" &&
    !data?.last_error;

  return (
    <Shell>
      <div className="topbar">
        <div>
          <div className="eyebrow">Market command center</div>
          <h1>Today’s decision layer</h1>
        </div>

        <div className="status">
          <span className="dot" />
          Execution disabled by design
        </div>
      </div>

      <div className="grid cols4">
        <div className="card">
          <h3>Top signal</h3>

          <div className="big">{action}</div>

          <div className="small">
            Confidence:{" "}
            {signal?.confidence != null
              ? `${signal.confidence.toFixed(1)}%`
              : "—"}
          </div>
        </div>

        <div className="card">
          <h3>Data health</h3>

          <div className="big">
            {loading ? "..." : dataHealthy ? "LIVE" : "WAIT"}
          </div>

          <div className="small">
            {data?.cycle_count ?? 0} analysis cycle
            {data?.cycle_count === 1 ? "" : "s"} completed
          </div>
        </div>

        <div className="card">
          <h3>Risk state</h3>

          <div className="big">
            {risk?.approved ? "APPROVED" : finalAction}
          </div>

          <div className="small">
            {risk?.assessment?.estimated_trade_risk != null
              ? `Risk ₹${Number(
                  risk.assessment.estimated_trade_risk
                ).toFixed(2)}`
              : "No approved trade"}
          </div>
        </div>

        <div className="card">
          <h3>Pipeline</h3>

          <div className="big">
            {data?.running ? "RUNNING" : "STOPPED"}
          </div>

          <div className="small">
            Refresh: {data?.interval_seconds ?? 30}s
          </div>
        </div>
      </div>

      <div className="section">
        <h2 className="section-title">Live decision</h2>

        <div className="grid cols3">
          <div className="card">
            <h3>Direction</h3>
            <div>{signal?.direction || "NONE"}</div>
            <div className="small">
              Score:{" "}
              {signal?.score != null
                ? Number(signal.score).toFixed(2)
                : "—"}
            </div>
          </div>

          <div className="card">
            <h3>Risk decision</h3>
            <div>{finalAction}</div>
            <div className="small">
              {risk?.assessment?.reason_codes?.length
                ? risk.assessment.reason_codes.join(", ")
                : "No risk blockers"}
            </div>
          </div>

          <div className="card">
            <h3>Market gate</h3>
            <div>
              {signal?.live_signal_gate?.passed ? "PASS" : "WAIT"}
            </div>
            <div className="small">
              {signal?.live_signal_gate?.reasons?.length
                ? signal.live_signal_gate.reasons.join(", ")
                : "No gate failures"}
            </div>
          </div>
        </div>
      </div>

      <div className="section">
        <h2 className="section-title">Pipeline</h2>

        <div className="grid cols3">
          {[
            "Data validation",
            "Market structure",
            "Options engine",
            "Strategy + ML",
            "Agent debate",
            "Risk + ranking",
          ].map((x, i) => (
            <div className="card" key={x}>
              <h3>0{i + 1}</h3>
              <div>{x}</div>
              <div className="small">
                Deterministic source of truth where applicable.
              </div>
            </div>
          ))}
        </div>
      </div>

      <div className="section note">
        {error
          ? `Backend connection error: ${error}`
          : data?.last_error
            ? `Pipeline error: ${data.last_error}`
            : data?.last_result?.timestamp
              ? `Last update: ${data.last_result.timestamp}`
              : "Waiting for the realtime pipeline to produce its first market result."}
      </div>
    </Shell>
  );
}
